53    /// 获取每一行的每一个字的元组数组
54    /// (start: String, length: String) (start: 开始时间, length: 持续时间)

57    static func wordTimes(lrc: String) -> [[(start: Double, length: Double)]] {
58        let separatedBy = lrc.contains("\r\n") ? "\r\n" : "\n"
59        //获取字符串：从[offset:0]后面的字符开始
60        let lrcs = String(lrc[lrc.range(of: "[offset:0]")!.upperBound...])
61        //把krc歌词每个单词的time的最后一个,0 去掉
62        let newLrc = lrcs.replacingOccurrences(of: ",0>", with: ">")
63        let lrcArray = newLrc.components(separatedBy: separatedBy)
64
65        var timeArray = [[(start: Double, length: Double)]]()
66        for lineLrc in lrcArray {
67            if lineLrc.count == 0 {
68                continue
69            }
70            // [669,850]<0,200>那<200,200>时<400,250>的<650,200>光
71            var oneLineArray = [(start: Double, length: Double)]()
72            let sub = String(lineLrc[lineLrc.range(of: "]")!.upperBound...])
73            //把sub按>分成数组
74            let wordArray = sub.components(separatedBy: ">")
75            for word in wordArray {
76                if !word.contains("<") {
77                    continue
78                }
79                //取出每个单词的开始时间
80                let arr = String(word[word.range(of: "<")!.upperBound...])
81                let start = Double(arr.components(separatedBy: ",")[0])!
82                let length = Double(arr.components(separatedBy: ",")[1])!
83                oneLineArray.append((start, length))
84            }
85            timeArray.append(oneLineArray)
86        }
87        return timeArray
88    }

129            // [669,850]<0,200>那<200,200>时<400,250>的<650,200>光
130            var lineArr = [String]()
131            let newArr = lineLrc.components(separatedBy: ">")
132            for (id, value) in newArr.enumerated() {
133                if id == 0 {
134                    continue
135                }
136                var s = ""
137                if id == (newArr.count-1) {
138                    s = String(value[value.startIndex...])
139                }else {
140                    s = String(value[value.startIndex ..< value.range(of: "<")!.lowerBound])
141                }
142                lineArr.append(s)
143            }
144            wordArr.append(lineArr)
145        }

151    /// 获取每一行的每一个字的元组数组
152    /// (start: String, length: String) (start: 开始时间, length: 持续时间)
153    /// - Parameter lrc: hrcs歌词
154    /// - Returns: 每一行的每一个字的元组数组
155    static func hrcs_wordTimes(lrc: String) -> [[(start: Double, length: Double)]] {
156        let separatedBy = lrc.contains("\r\n") ? "\r\n" : "\n"
157        //获取字符串：从 haplayer.lrc 后面的字符开始
158        let lrcs = String(lrc[lrc.range(of: "haplayer.lrc")!.lowerBound...])
159        let lrcArray = lrcs.components(separatedBy: separatedBy)
160
161        var timeArray = [[(start: Double, length: Double)]]()
162        for lineLrc in lrcArray {
163            if lineLrc.count == 0 {
164                continue
165            }
166            // haplayer.lrc('<5100,11470>','<问><心><( ><主><题><曲><)>','<1000,1010,710,910,910,900,930>');
167            let wordArray = lineLrc.components(separatedBy: ",'")
168            let arr = String(wordArray[2][wordArray[2].range(of: "<")!.upperBound ..< wordArray[2].range(of: ">")!.lowerBound])
169            let arrs = arr.components(separatedBy: ",").map { (Double(0), Double($0)!) }
170            timeArray.append(arrs)
171        }
172        return timeArray
173    }
